<?php
function boundary(int $quantity): int {
    if ($quantity >= 3) {
        return 1;
    }
    return 0;
}
function complement(int $count, bool $priority): int {
    if (!($count < 3 || !$priority)) {
        return 7;
    } else {
        return 9;
    }
}
function early(int $amount): bool {
    if ($amount < 0) {
        return false;
    }
    return true;
}
function omission(int $tier): int {
    if ($tier >= 10) {
        return 1;
    } elseif ($tier >= 5) {
        return 2;
    }
    return 3;
}
function relations(int $first, int $second): bool {
    return $first < $second;
}
function elseOrder(int $limit): int {
    if ($limit >= 10) {
        return 1;
    }
    if ($limit >= 5) {
        return 2;
    }
    return 3;
}
function category(int $tier): int {
    if ($tier === 1 || $tier === 2) {
        return 5;
    }
    return 9;
}
